-- | Profunctor optics for 'Text'.
--
-- @
-- \-\- convert strict Text to lazy Text
-- view lazy myText
--
-- \-\- pack a String into Text
-- review packed "hello"
-- @

    short,

    -- * Lazy\/strict iso
    lazy,

    -- * Packing
    packed,

    -- * UTF-8 encoding

) where
